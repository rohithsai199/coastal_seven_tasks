#basic calculator

num1 = float(input("enter first number: "))
op = input("enter operator (+, -, *, /): ")
num2 = float(input("enter second number: "))
if op == "+":
    print("result:", num1 + num2)
elif op == "-":
    print("result:", num1 - num2)
elif op == "*":
    print("result:", num1 * num2)
elif op == "/":
    if num2 == 0:
        print("Error: Division by zero")
    else:
        print("Result:", num1 / num2)
else:
    print("Invalid operator")

#list operations
#append
#extend
#clear
#index
#pop
#count
num = [10, 20, 30, 40]
op = input("Enter operation (append, clear, extend, pop, count, index): ")

# 1. append: Adds an item to the very end of the list
if op == "append":
    num.append(50)
    print("New list:", num)

# 2. clear: Removes everything, leaving an empty list
if op == "clear":
    num.clear()
    print("New list:", num)

# 3. extend: Adds multiple items to the end of the list
if op == "extend":
    num.extend([50, 60])
    print("New list:", num)

# 4. pop: Removes the last item from the list
if op == "pop":
    num.pop()
    print("New list:", num)

# 5. count: Tells you how many times a number appears
if op == "count":
    result = num.count(20)
    print("The number 20 appears this many times:", result)

# 6. index: Tells you the position (index) of a number
if op == "index":
    result = num.index(30)
    print("The number 30 is at index:", result)

# dictionary operations

dici = {"a": 10, "b": 20, "c": 30}
op = input("Enter operation (update, get, pop, clear, keys, values): ")

# 1. update: Adds a new key-value pair to the dictionary
if op == "update":
    dici.update({"d": 40})
    print("New dictionary:", dici)

# 2. get: Safely looks up the value of a specific key
if op == "get":
    result = dici.get("a")
    print("The value for 'a' is:", result)

# 3. pop: Removes a specific key and its value
if op == "pop":
    dici.pop("b")
    print("New dictionary after popping 'b':", dici)

# 4. clear: Removes everything, leaving an empty dictionary
if op == "clear":
    dici.clear()
    print("New dictionary:", dici)

# 5. keys: Shows all the keys in the dictionary
if op == "keys":
    result = dici.keys()
    print("The keys are:", result)

# 6. values: Shows all the values in the dictionary
if op == "values":
    result = dici.values()
    print("The values are:", result)
  

#grade calculator

score1 = float(input("Enter score for subject 1: "))
score2 = float(input("Enter score for subject 2: "))
score3 = float(input("Enter score for subject 3: "))

total = score1 + score2 + score3
average = total / 3
if average >= 90:
    grade = "A"
elif average >= 80:
    grade = "B"
elif average >= 70:
    grade = "C"
elif average >= 60:
    grade = "D"
else:
    grade = "F"
print("Total Score:", total)

#simple function operations

def is_prime(num):
  if num < 2:
    return False
  
  for i in range(2, int(num**0.5) + 1):
    if num % i == 0:  
      return False
  return True

num = int(input("enter: "))
if is_prime(num):
  print("prime number")
else:
  print("not a prime number")
